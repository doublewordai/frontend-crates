def run(parser_name, text, chunk):
    parser = FunctionCallParser(TOOLS, parser_name)
    events = []
    for i in range(0, len(text), chunk):
        piece = text[i:i + chunk]
        normal, calls = parser.parse_stream_chunk(piece)
        if normal:
            events.append(("C", normal))
        for c in calls:
            events.append(("T", c.name, c.parameters))
    return events


def summarize(events):
    return "".join(e[0] for e in events)


for name, text in CASES.items():
    print("=" * 78)
    print(f"{name}   (len={len(text)})")
    base = run(name, text, 1)
    print("  chunk=1: ", summarize(base), [(e[0], e[1:] if e[0] == 'T' else e[1]) for e in base])
    for chunk in (2, 3, 4, 6, 10, 20, 10 ** 6):
        ev = run(name, text, chunk)
        s = summarize(ev)
        flag = "" if s == summarize(base) else "   <-- ORDER MISMATCH"

        if flag:
